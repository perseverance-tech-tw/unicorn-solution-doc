105 inline void initAudioOnce(const std::string &customFile = "")
106 {
107 using namespace detail;
108 if (ready())
109 return;
110
111 if (!customFile.empty())
112 soundFile() = customFile;
113
114 if (soundFile().empty())
115 {
116 if (std::filesystem::exists("/usr/share/sounds/freedesktop/stereo/bell.oga"))
117 soundFile() = "/usr/share/sounds/freedesktop/stereo/bell.oga";
118 else if (std::filesystem::exists("/usr/share/sounds/alsa/Noise.wav"))
119 soundFile() = "/usr/share/sounds/alsa/Noise.wav"; // cadangan
120 }
121
122 std::string ext;
123 if (auto pos = soundFile().rfind('.'); pos != std::string::npos)
124 ext = soundFile().substr(pos + 1);
125
126 if ((ext == "ogg" || ext == "oga") && which("paplay"))
127 {
128 player() = "paplay";
129 cmdTemplate() = "paplay '%FILE%' >/dev/null 2>&1";
130 }
131 else if (which("aplay"))
132 {
133 player() = "aplay";
134 cmdTemplate() = "aplay -q '%FILE%' >/dev/null 2>&1";
135 }
136 else if (which("paplay"))
137 {
138 player() = "paplay";
139 cmdTemplate() = "paplay '%FILE%' >/dev/null 2>&1";
140 }
141 else
142 {
143 player().clear();
144 }
145
146 std::cerr << "[BEEPER] init: file=" << (soundFile().empty() ? "(none)" : soundFile())
147 << " player=" << (player().empty() ? "(BEL)" : player()) << "\n";
148 ready() = true;

195 inline void stop(const std::string &key)
196 {
197 using namespace detail;
198 std::unique_lock<std::mutex> lk(mtx());
199 auto itR = running().find(key);
200 auto itT = threads().find(key);
201 if (itR == running().end())
202 return;
203
204 itR->second.store(false, std::memory_order_release);
205 std::thread th;
206 if (itT != threads().end())
207 {
208 th = std::move(itT->second);
209 threads().erase(itT);
210 }
211 running().erase(itR);
212 lk.unlock();
213 if (th.joinable())
214 th.join();
215 }

270 inline void update(const std::string &key, VRUDetector::BeepPattern pat, const Config &cfg = {})
271 {
272 using namespace detail;
273
274 if (pat != VRUDetector::BeepPattern::BEEP_3_LOOP)
275 {
276 stop(key);
277 }
278
279 switch (pat)
280 {
281 case VRUDetector::BeepPattern::NONE:
282 return;
283
284 case VRUDetector::BeepPattern::BEEP_2_ONCE:
285 {
286 std::thread([cfg]
287 {
288 beep_once();
289 std::this_thread::sleep_for(std::chrono::milliseconds(cfg.betweenBeepsMs));
290 beep_once(); })
291 .detach();
292 return;
293 }
294
295 case VRUDetector::BeepPattern::BEEP_3_LOOP:
296 {
297 std::lock_guard<std::mutex> lk(mtx());
298 if (running().count(key) && running()[key].load(std::memory_order_acquire))
299 return;
300
301 running()[key].store(true, std::memory_order_release);
302 threads()[key] = std::thread([key, cfg]
303 {
304 while (true) {
305 {
306 std::lock_guard<std::mutex> lk(mtx());
307 if (!running()[key].load(std::memory_order_acquire)) break;
308 }
309 beep_once();
310 std::this_thread::sleep_for(std::chrono::milliseconds(cfg.betweenBeepsMs));
311 beep_once();
312 std::this_thread::sleep_for(std::chrono::milliseconds(cfg.betweenBeepsMs));
313 beep_once();
314
315 std::this_thread::sleep_for(std::chrono::milliseconds(cfg.loopPeriodMs));
316 } });
317 return;
318 }
